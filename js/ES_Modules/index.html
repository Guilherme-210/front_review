Módulos com ES Modules

O ES Modules (ECMAScript Modules) é um sistema de módulos nativo do JavaScript que permite dividir o código em arquivos reutilizáveis e organizados. Ele usa as palavras-chave export e import para compartilhar e carregar funcionalidades entre arquivos.

Para usar ES Modules no HTML, você deve definir a chamada da tag <script> com o type="module"

Exemplo sem a utilização dos ES Modules

Vamos imaginar que na nossa aplicação temos um arquivo javascript com várias funções e que utilizamos essas funções várias vezes: 

// script1.js

function label(attributes) {
  const element = document.createElement('label')
  Object.assign(element, attributes)
  return element
}

function input(attributes) {
  const element = document.createElement('input')
  Object.assign(element, attributes)
  return element
}

function br() {
  const element = document.createElement('br')
  return element
}

console.log(label({ for: 'fullname', textContent: 'Nome Completo'}))
console.log(input({ id: 'fullname', name: 'fullname', placeholder: 'Digite seu nome completo...'}))
console.log(br())
Exemplo utilizando os ES Modules

Vamos modularizar nosso código javascript utilizando os ES Modules: 

// index.html
            
<script src="scripts/script2.js" type="module"></script>
// script2.js

import { name, label, input, br } from "./functions.js"

console.log(name)
console.log(label({ for: "fullname", textContent: "Nome Completo" }))
console.log(
  input({
    id: "fullname",
    name: "fullname",
    placeholder: "Digite seu nome completo...",
  })
)
console.log(br())
// functions.js

export const name = "Guilherme"

export function label(attributes) {
  const element = document.createElement("label")
  Object.assign(element, attributes)
  return element
}

export function input(attributes) {
  const element = document.createElement("input")
  Object.assign(element, attributes)
  return element
}

export function br() {
  const element = document.createElement("br")
  return element
}
